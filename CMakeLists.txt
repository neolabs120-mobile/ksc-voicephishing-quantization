cmake_minimum_required(VERSION 3.20)
project(ksc_gguf_evaluation LANGUAGES C CXX)

# 연구 당시 llama.cpp를 외부 경로로 연결해 모델·서드파티 소스를 저장소에 복제하지 않는다.
set(LLAMA_CPP_DIR "" CACHE PATH "llama.cpp b9010 checkout")
if(NOT EXISTS "${LLAMA_CPP_DIR}/include/llama.h")
    message(FATAL_ERROR "Set LLAMA_CPP_DIR to the pinned llama.cpp b9010 checkout")
endif()
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_CPP_DIR}" llama.cpp)
add_executable(gguf_eval scripts/gguf_eval.cpp)
target_compile_features(gguf_eval PRIVATE cxx_std_17)
target_include_directories(gguf_eval PRIVATE "${LLAMA_CPP_DIR}/vendor/nlohmann")
target_link_libraries(gguf_eval PRIVATE llama ggml ggml-base)
set_target_properties(gguf_eval PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
